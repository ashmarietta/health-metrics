# run again for variable consistency
import csv
import xml.etree.ElementTree as ET
from collections import defaultdict
from datetime import datetime

INPUT = "export.xml" # original export
PERIODS = [
    ("Off-Season Softball", "2025-08-26", "2026-02-05"),
    ("In-Season Softball", "2026-02-06", "2026-05-06"),
    ("Half Marathon Running", "2026-06-08", "2026-09-19"),
]
# daily total categories
SUM_METRICS = {
    "ActiveEnergyBurned", "BasalEnergyBurned", "DistanceWalkingRunning",
    "StepCount", "AppleExerciseTime", "DistanceCycling",
}
# daily mean / min / max / count categories
STAT_METRICS = {
    "HeartRate", "HeartRateVariabilitySDNN", "RespiratoryRate", "PhysicalEffort",
    "RunningSpeed", "RunningPower", "RunningVerticalOscillation",
    "RunningGroundContactTime", "RunningStrideLength", "WalkingStepLength",
    "WalkingAsymmetryPercentage", "VO2Max", "AppleSleepingWristTemperature",
    "HeartRateRecoveryOneMinute", "RestingHeartRate", "WalkingHeartRateAverage",
    "AppleWalkingSteadiness", "SixMinuteWalkTestDistance", "OxygenSaturation",
    "BodyMass",
}
# daily event counts
COUNT_METRICS = {"LowHeartRateEvent"}
KEEP_WORKOUTS = {
    "HKWorkoutActivityTypeWalking", "HKWorkoutActivityTypeSoftball",
    "HKWorkoutActivityTypeRunning", "HKWorkoutActivityTypeTraditionalStrengthTraining",
}

def short(t):
    for p in ("HKQuantityTypeIdentifier", "HKCategoryTypeIdentifier", "HKDataType"):
        t = t.replace(p, "")
    return t

def period_of(day):
    for name, lo, hi in PERIODS:
        if lo <= day <= hi:
            return name
    return None

def is_apple_watch_source(source):
    return "Apple" in source and "Watch" in source

# (day, metric) to [count, sum, min, max]
acc = defaultdict(lambda: [0, 0.0, float("inf"), float("-inf")])
sleep_duration = defaultdict(float)
hrv_file = open("hrv_readings.csv", "w", newline="", encoding="utf-8")
hrv_w = csv.writer(hrv_file)
hrv_w.writerow(["period", "date", "start", "end", "value", "unit", "motion_context", "source"])

wk_file = open("workouts.csv", "w", newline="", encoding="utf-8")
wk_w = csv.writer(wk_file)
wk_w.writerow(["period", "date", "type", "start", "end", "duration", "duration_unit",
               "distance", "distance_unit", "energy", "energy_unit", "source"])

context = ET.iterparse(INPUT, events=("start", "end"))
_, root = next(context)
seen = 0

for event, elem in context:
    if event != "end" or elem.tag not in ("Record", "Workout"):
        continue
    seen += 1
    day = elem.get("startDate", "")[:10]
    period = period_of(day)

    source = elem.get("sourceName", "")
    if period and is_apple_watch_source(source):
        if elem.tag == "Record":
            metric = short(elem.get("type", ""))
            if metric == "SleepAnalysis" and elem.get("value", "").startswith(
                "HKCategoryValueSleepAnalysisAsleep"
            ):
                start = datetime.strptime(elem.get("startDate"), "%Y-%m-%d %H:%M:%S %z")
                end = datetime.strptime(elem.get("endDate"), "%Y-%m-%d %H:%M:%S %z")
                sleep_duration[day] += (end - start).total_seconds() / 3600
            elif metric in SUM_METRICS or metric in STAT_METRICS or metric in COUNT_METRICS:
                a = acc[(day, metric)]
                if metric in COUNT_METRICS:
                    a[0] += 1
                else:
                    try:
                        v = float(elem.get("value"))
                    except (TypeError, ValueError):
                        v = None
                    if v is not None:
                        a[0] += 1
                        a[1] += v
                        a[2] = min(a[2], v)
                        a[3] = max(a[3], v)
                if metric == "HeartRateVariabilitySDNN":
                    ctx = ""
                    for m in elem.iter("MetadataEntry"):
                        if m.get("key") == "HKMetadataKeyHeartRateMotionContext":
                            ctx = m.get("value")
                    hrv_w.writerow([period, day, elem.get("startDate"), elem.get("endDate"),
                                    elem.get("value"), elem.get("unit"), ctx,
                                    source])
        else:
            if elem.get("workoutActivityType") in KEEP_WORKOUTS:
                wk_w.writerow([period, day, elem.get("workoutActivityType"),
                               elem.get("startDate"), elem.get("endDate"),
                               elem.get("duration"), elem.get("durationUnit"),
                               elem.get("totalDistance"), elem.get("totalDistanceUnit"),
                               elem.get("totalEnergyBurned"), elem.get("totalEnergyBurnedUnit"),
                               source])
    elem.clear()
    if seen % 50000 == 0:
        root.clear()

hrv_file.close()
wk_file.close()

# daily table with one row per date
days = sorted({d for d, _ in acc})
metrics = sorted({m for _, m in acc})
header = ["date", "period"]
for m in metrics:
    if m in SUM_METRICS:
        header.append(f"{m}_sum")
    elif m in COUNT_METRICS:
        header.append(f"{m}_count")
    else:
        header += [f"{m}_mean", f"{m}_min", f"{m}_max", f"{m}_n"]

with open("daily_summary.csv", "w", newline="", encoding="utf-8") as f:
    w = csv.writer(f)
    w.writerow(header + ["SleepDuration_hours"])
    for d in days:
        row = [d, period_of(d)]
        for m in metrics:
            a = acc.get((d, m))
            if m in SUM_METRICS:
                row.append(round(a[1], 3) if a and a[0] else "")
            elif m in COUNT_METRICS:
                row.append(a[0] if a else "")
            elif a and a[0]:
                row += [round(a[1] / a[0], 3), a[2], a[3], a[0]]
            else:
                row += ["", "", "", ""]
        row.append(round(sleep_duration.get(d, 0), 3))
        w.writerow(row)

# quick hrv check by period
print(f"Scanned {seen:,} records/workouts. Wrote daily_summary.csv, hrv_readings.csv, workouts.csv\n")
for name, lo, hi in PERIODS:
    vals = [acc[(d, "HeartRateVariabilitySDNN")] for d in days
            if lo <= d <= hi and acc.get((d, "HeartRateVariabilitySDNN"), [0])[0]]
    if vals:
        daily_means = [a[1] / a[0] for a in vals]
        print(f"{name} ({lo} to {hi}): {len(vals)} days with HRV, "
              f"mean of daily HRV = {sum(daily_means)/len(daily_means):.1f} ms")

import pandas as pd
df = pd.read_csv("daily_summary.csv")
df1 = pd.read_csv("hrv_readings.csv")
df2 = pd.read_csv("workouts.csv")

df1["value"] = pd.to_numeric(df1["value"], errors="coerce")
hrv_daily = (df1.groupby("date", as_index=False)
           .agg(HRV_readings=("value", "mean")))

df2["duration"] = pd.to_numeric(df2["duration"], errors="coerce")
df2["distance"] = pd.to_numeric(df2["distance"], errors="coerce")
df2["energy"] = pd.to_numeric(df2["energy"], errors="coerce")
workouts_daily = (df2.groupby("date", as_index=False)
               .agg(workout_count=("type", "size"),
                   workout_duration=("duration", "sum"),
                   workout_distance=("distance", "sum"),
                   workout_energy=("energy", "sum")))

workout_type_names = {
    "HKWorkoutActivityTypeRunning": "running",
    "HKWorkoutActivityTypeSoftball": "softball",
    "HKWorkoutActivityTypeWalking": "walking",
    "HKWorkoutActivityTypeTraditionalStrengthTraining": "strength",
}
typed_workouts = df2[df2["type"].isin(workout_type_names)].copy()
typed_workouts["workout_type"] = typed_workouts["type"].map(workout_type_names)
typed_workouts_daily = (
    typed_workouts.groupby(["date", "workout_type"])
    .agg(
        workout_count=("type", "size"),
        workout_duration=("duration", "sum"),
    )
    .unstack("workout_type", fill_value=0)
)
typed_workouts_daily.columns = [
    f"{metric}_{workout_type}"
    for metric, workout_type in typed_workouts_daily.columns
]
typed_workouts_daily = typed_workouts_daily.reset_index()

dfall = pd.merge(df, hrv_daily, how="left", on="date")
dfall = pd.merge(dfall, workouts_daily, how="left", on="date")
dfall = pd.merge(dfall, typed_workouts_daily, how="left", on="date")
typed_workout_columns = [
    f"{metric}_{workout_type}"
    for workout_type in workout_type_names.values()
    for metric in ("workout_count", "workout_duration")
]
dfall[typed_workout_columns] = dfall[typed_workout_columns].fillna(0)
dfall['date'] = pd.to_datetime(dfall['date'])
dfall.set_index('date', inplace=True)

DESCRIBE METRICS & WHY I CHOSE THEM

HRV is a measure of the variation in time between heartbeats in milliseconds, and it is often used as an indicator of recovery and autonomic nervous system balance. Higher HRV is generally associated with better cardiovascular fitness and recovery status.

Resting heart rate is the number of heartbeats per minute (bpm) while at rest, and lower values are typically indicative of better cardiovascular fitness. Healthy adults usually have a resting heart rate between 60 and 100 bpm, with athletes often having lower rates.

VO2 max is the maximum rate of oxygen consumption during intense exercise, and higher values indicate better aerobic fitness and endurance capacity.

I chose these metrics because they are commonly used indicators of cardiovascular health, fitness, and recovery. They provide insights into how the body responds to training loads and recovery periods, which is particularly relevant for athletes transitioning between different sports or training regimens. I am fascinated by human lifespan and healthspan, and these are common markers for tracking the overall longevity of humans. Naturally, I wanted to start keeping a log of my own, especially as my fitness journey has taken me from a softball player to a distance runner.
